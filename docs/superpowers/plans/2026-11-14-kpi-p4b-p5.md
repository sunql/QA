```python
"""measure → SQL 编译器（P4b）。

只读结构化列，不嗅探 formula 文本。任一输入不足以确定性编译 ⇒ fail-closed，
上层降级 Tier B；绝不发半成品 SQL。
"""

COMPILED_* = "compiled_..."   # 全部原因码常量

@dataclass(frozen=True)
class CompiledSql: ...

_QUOTE = {"oracle": '"%s"', "postgresql": '"%s"', "mysql": "`%s`"}
_AGG = {"SUM": "SUM", "COUNT": "COUNT", "AVG": "AVG", "COUNT_DISTINCT": "COUNT(DISTINCT {expr})"}


def compileMetricSql(metric, *, sources, classes, joins, datasourceType, schemaPrefix=None, timeScope=None) -> CompiledSql:
    # 1) 方言 / metric_type / measure 校验
    # 2) 收集 measure 表达式：expr 缺失且 agg=COUNT ⇒ "*"；COUNT_DISTINCT 必须有 expr
    # 3) 片段守卫：_assertSafeFragment
    # 4) 源类解析：measure.class_id → class → source_table；无 class_id 时回退唯一源类
    # 5) 跨表：class_id 图 + ontology_join 边，BFS 找路径；按路径渲染 INNER JOIN
    # 6) 表达式按 metric_type/aggregation_order 拼装；measure.filter 用 CASE WHEN
    # 7) WHERE = [global filter, time predicate]
    # 8) _assert_read_only(sql)（import 自 business_db_pool）
```

实现约束：

- 引号/表名前缀：表名用 `_bakeTableName` 同义逻辑（`f"{schemaPrefix}.{table}"`，schema 只允许 `[A-Za-z_][A-Za-z0-9_]*`，否则 fail `compiled_invalid_fragment`）。
- `_assertSafeFragment`：空则视为 None；拒绝 `; -- /* */`，括号/引号不平衡，DML 关键词 `\b(insert|update|delete|drop|alter|truncate|grant|create)\b`（忽略大小写）。
- `raw measure filter`：`CASE WHEN {filter} THEN {expr} END`，COUNT(*) 同理 `CASE WHEN {filter} THEN 1 END`。
- 跨表 JOIN 的列名：`"<table>"."<col>"`；join_type 只接受 `INNER|LEFT`（其他 ⇒ `compiled_join_path_missing`）。
- `grain_keys` 非空时 `warnings=("grain_keys_ignored",)`。
- 最终 `_assert_read_only(sql)` 失败 ⇒ `compiled_not_read_only`。

```python
async def loadAndCompileMetricSql(
    session: AsyncSession,
    metric: OntologyMetric,
    *,
    datasourceType: str,
    schemaPrefix: str | None = None,
    timeScope: Any | None = None,
) -> CompiledSql:
    sources = (await session.execute(select(OntologyMetricSource).where(...))).scalars().all()
    classIds = {...}
    classes = (await session.execute(select(OntologyClass).where(OntologyClass.id.in_(classIds)))).scalars().all()
    joins = (await session.execute(select(OntologyJoin).where(...))).scalars().all()
    return compileMetricSql(metric, sources=sources, classes=classes, joins=joins, ...)
```

```python
_KPI_TRANSITIONS = {
    "DRAFT": frozenset({"PUBLISHED"}),
    "PUBLISHED": frozenset({"DEPRECATED"}),

def assertKpiTransition(fromStatus: str, toStatus: str) -> None:
    src, dst = normalize(fromStatus), normalize(toStatus)
    if src == dst:
        return
    allowed = _KPI_TRANSITIONS.get(src, frozenset())
    if dst not in allowed:
        raise ConflictError(
            MSG_KPI_STATUS_TRANSITION_INVALID.format(src=src, dst=dst, allowed=...),

            # ConflictError 不接受 details；若基类不支持，用 detail 承载 JSON 字符串或扩展 ConflictError
        )
```

> class DomainError(Exception):

>     ) -> None:
>         super().__init__(message)
>         self.message = message
>         self.detail = detail
>         self.details = details
> ```
